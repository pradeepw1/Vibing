%%concept run
%%title Running your program
%%group Real work
%%blurb How the code you wrote actually gets to run.
%%docs compiler interpreter jcl jvm
%%explain
Some languages are run line by line by an interpreter. Others are compiled first: translated into something the machine can run.

Here's what you type for each one. On a mainframe you don't type a command at all: you submit a job, and it runs when its turn comes.
%%mainframe
JCL is a set of instructions for the operating system: which program to run, and which datasets to give it. Each `EXEC` is a step, and each `DD` connects a file to the program. You submit the job, it waits in a queue, and you read the result in SDSF. Compiling is a job too. Production work runs overnight under a scheduler, and step return codes decide what runs next.

%%concept database
%%title Talking to a database
%%group Real work
%%blurb Save data that outlives your program, and ask questions of it.
%%docs database sql sql-injection db2
%%explain
A database keeps data safe and lets you ask for exactly what you need. SQL is the language nearly all of them speak, and it looks the same from every language below.

One rule matters everywhere: never paste a person's input into SQL. Use a `?` or `:name` placeholder, so input can only ever be data. Pasting it in is how SQL injection attacks happen.
%%mainframe
In COBOL, SQL sits right inside the program between `EXEC SQL` and `END-EXEC`. A precompiler swaps it for real calls, and a BIND step prepares the plan. `:WS-NAME` is a host variable, your COBOL field. After every statement you check `SQLCODE`: 0 is fine, 100 means not found, and a negative number is an error.

%%lang python
%%note `sqlite3` comes with Python. This one lives in memory and needs no setup.
%%code
import sqlite3

db = sqlite3.connect(":memory:")
db.execute("CREATE TABLE customer (name TEXT, balance REAL)")
db.execute("INSERT INTO customer VALUES (?, ?)", ("Asha", 150.0))

query = "SELECT name, balance FROM customer WHERE balance > ?"
for row in db.execute(query, (100,)):
    print(row)
%%output
('Asha', 150.0)

%%lang java
%%note JDBC is the standard way in. `?` keeps input safe.
%%run compile
%%code
import java.sql.DriverManager;

public class Main {
    public static void main(String[] args) throws Exception {
        String sql = "SELECT name, balance FROM customer WHERE balance > ?";
        try (var db = DriverManager.getConnection("jdbc:postgresql://localhost/shop");
             var query = db.prepareStatement(sql)) {
            query.setInt(1, 100);
            var rows = query.executeQuery();
            while (rows.next()) {
                System.out.println(rows.getString("name"));
            }
        }
    }
}

           END-EVALUATE.

%%concept async
%%title Doing several things at once
%%group Real work
%%blurb Start slow jobs together, instead of waiting for each in turn.
%%docs async thread
%%explain
Fetching from the network is slow. If you wait for one thing, then the next, the whole app freezes up. Async code lets you start the work, get on with other things, and pick up the result when it's ready.

`await` means "pause this function until the answer is here". These examples start two jobs at once, then collect both answers.
%%mainframe
COBOL has no async. Batch programs run top to bottom on one track. Speed comes from outside the program: the scheduler runs jobs side by side, and big files are split so several jobs can each take a part. Online, CICS runs many transactions at once, each a separate short task. Parallel Sysplex and Db2 data sharing spread that over several machines.

%%lang python
%%note `gather` runs them together and returns the answers in order.
%%code
import asyncio

async def fetch_price(name, delay):
    await asyncio.sleep(delay)
    return f"{name}: ok"

async def main():
    results = await asyncio.gather(
        fetch_price("apple", 0.2),
        fetch_price("pear", 0.1),
    )
    print(results)

asyncio.run(main())
%%output
['apple: ok', 'pear: ok']

%%lang swift
%%note `async let` starts the work. `await` collects it.
%%code
func fetchPrice(_ name: String, delay: UInt64) async -> String {
    try? await Task.sleep(nanoseconds: delay)
    return "\(name): ok"
}

async let apple = fetchPrice("apple", delay: 200_000_000)
async let pear = fetchPrice("pear", delay: 100_000_000)
print(await [apple, pear])
%%output
["apple: ok", "pear: ok"]

%%lang java
%%note `supplyAsync` runs work on another thread. `join` waits for the answer.
%%code
import java.util.concurrent.CompletableFuture;

public class Main {
    static String fetchPrice(String name, long delay) {
        try {
            Thread.sleep(delay);
        } catch (InterruptedException e) {
            throw new RuntimeException(e);
        }
        return name + ": ok";
    }

    public static void main(String[] args) {
        var apple = CompletableFuture.supplyAsync(() -> fetchPrice("apple", 200));
        var pear = CompletableFuture.supplyAsync(() -> fetchPrice("pear", 100));
        System.out.println(apple.join() + ", " + pear.join());
    }
}
%%output
apple: ok, pear: ok

%%lang cobol
%%skip COBOL runs one step at a time. Doing things at once is the scheduler's job.

%%lang js
%%note JavaScript is built around this. `Promise.all` waits for every answer.
%%code
const wait = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

async function fetchPrice(name, delay) {
  await wait(delay);
  return `${name}: ok`;
}

const results = await Promise.all([
  fetchPrice("apple", 200),
  fetchPrice("pear", 100),
]);
console.log(results.join(", "));
%%output
apple: ok, pear: ok

%%lang ts
%%note `Promise<string>` means "a string, later".
%%code
const wait = (ms: number): Promise<void> =>
  new Promise((resolve) => setTimeout(resolve, ms));

async function fetchPrice(name: string, delay: number): Promise<string> {
  await wait(delay);
  return `${name}: ok`;
}

const results = await Promise.all([
  fetchPrice("apple", 200),
  fetchPrice("pear", 100),
]);
console.log(results.join(", "));
%%output
apple: ok, pear: ok
